                <label>Название клуба / Локация</label>
                <input type="text" id="clubName" placeholder="Например: Colizeum" value="CyberZone Arena">

            Загрузка данных...

        <button class="action-btn btn-secondary" onclick="toggleExport()">📥 Экспорт / Импорт данных</button>

<script>
    // Инициализация Telegram WebApp
    const tg = window.Telegram.WebApp;
    tg.expand(); // Раскрыть на весь экран
    
    // Получаем ID пользователя и имя
    const user = tg.initDataUnsafe.user;
    const userId = user ? user.id : 'guest_' + Math.floor(Math.random() * 10000);
    const userName = user ? (user.first_name || 'Игрок') : 'Гость';

    // Структура данных (по умолчанию пустая или из localStorage)
    let appData = {
        club: "CyberZone Arena",
        votes: {}, // { userId: [1, 5, 12] } - массив выбранных дней
        lockedDate: null // Если дата зафиксирована
    };

    // Загрузка из LocalStorage при старте
    const savedData = localStorage.getItem('pc_club_data');
    if (savedData) {
        try { appData = JSON.parse(savedData); } catch(e) {}
    }

    // Восстанавливаем название клуба в поле
    document.getElementById('clubName').value = appData.club;

    // Генерация календаря (Ноябрь 2026 - 30 дней)
    const calendarEl = document.getElementById('calendar');
    const myVotes = appData.votes[userId] || [];

    for (let i = 1; i <= 30; i++) {
        const dayEl = document.createElement('div');
        dayEl.className = 'day';
        dayEl.innerText = i;
        
        // Если этот день выбран мной
        if (myVotes.includes(i)) {
            dayEl.classList.add('selected');
        }
        
        // Если день зафиксирован админом
        if (appData.lockedDate === i) {
            dayEl.classList.add('locked');
            dayEl.innerHTML += ' 🔒';
        }

        // Клик по дню
        dayEl.onclick = () => toggleDay(i, dayEl);
        calendarEl.appendChild(dayEl);
    }

    // Функция переключения дня
    function toggleDay(day, element) {
        if (appData.lockedDate) return; // Нельзя менять, если зафиксировано

        let currentVotes = appData.votes[userId] || [];
        
        if (currentVotes.includes(day)) {
            currentVotes = currentVotes.filter(d => d !== day);
            element.classList.remove('selected');
        } else {
            currentVotes.push(day);
            element.classList.add('selected');
        }
        
        appData.votes[userId] = currentVotes;
        saveToLocal();
        updateStatus();
        
        // Вибрация (Haptic Feedback) для ощущения приложения
        if(tg.HapticFeedback) tg.HapticFeedback.impactOccurred('light');
    }

    // Сохранение выбора
    function saveVotes() {
        appData.club = document.getElementById('clubName').value;
        saveToLocal();
        updateStatus();
        tg.showPopup({
            title: 'Готово!',
            message: 'Твой выбор сохранен локально. Скинь "Экспорт" друзьям, чтобы синхронизироваться.',
            buttons: [{type: 'ok'}]
        });
    }

    // Сохранение в память браузера
    function saveToLocal() {
        localStorage.setItem('pc_club_data', JSON.stringify(appData));
    }

    // Обновление статуса (кто сколько голосов набрал)
    function updateStatus() {
        const counts = {};
        // Считаем голоса
        Object.values(appData.votes).forEach(days => {
            days.forEach(d => {
                counts[d] = (counts[d] || 0) + 1;
            });
        });

        // Находим топ день
        let topDay = 0, maxVotes = 0;
        for (const [day, count] of Object.entries(counts)) {
            if (count > maxVotes) {
                maxVotes = count;
                topDay = day;
            }
        }

        const statusEl = document.getElementById('status-display');
        statusEl.classList.remove('hidden');
        
        if (appData.lockedDate) {
            statusEl.innerHTML = `🔒 <b>КАТКА ЗАФИКСИРОВАНА!</b><br>Дата: ${appData.lockedDate} ноября<br>Клуб: ${appData.club}`;
            statusEl.style.borderColor = 'var(--accent)';
        } else if (topDay > 0) {
            statusEl.innerHTML = `🔥 <b>Лидер голосования:</b> ${topDay} ноября (${maxVotes} голосов)<br>Клуб: ${appData.club}`;
            statusEl.style.borderColor = 'var(--primary)';
        } else {
            statusEl.innerHTML = `📅 Голосование идет...<br>Клуб: ${appData.club}`;
        }
    }

    // Экспорт / Импорт
    function toggleExport() {
        const area = document.getElementById('export-area');
        const textarea = document.getElementById('json-data');
        area.classList.toggle('hidden');
        if (!area.classList.contains('hidden')) {
            textarea.value = JSON.stringify(appData);
        }
    }

    function importData() {
        try {
            const raw = document.getElementById('json-data').value;
            const imported = JSON.parse(raw);
            if (imported.votes) {
                // Объединяем голоса (чтобы не затереть свои, если они есть)
                appData = imported; 
                // Восстанавливаем свой текущий выбор, если он был, или принимаем новый
                saveToLocal();
                location.reload(); // Перезагружаем, чтобы обновить календарь
            }
        } catch (e) {
            alert('Ошибка формата данных!');
        }
    }

    // Первый запуск статуса
    updateStatus();

    // Настройка главной кнопки Telegram (опционально)
    tg.MainButton.setText("СКИНУТЬ СВОДКУ В ЧАТ");
    tg.MainButton.onClick(() => {
        const text = `🎮 Сбор в ${appData.club}\n📅 Топ дата: ${Object.keys(appData.votes).length > 0 ? 'есть голоса' : 'нет голосов'}\n👉 Открой приложение, чтобы проголосовать!`;
        tg.sendData(text); // Отправляет данные боту (если настроен) или просто закрывает
        // Для простоты копирования:
        navigator.clipboard.writeText(JSON.stringify(appData));
        tg.showAlert("Данные скопированы! Вставь их в чат, чтобы друзья импортировали.");
    });
    tg.MainButton.show();

</script>
